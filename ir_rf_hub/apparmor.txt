#include <tunables/global>

profile ir_rf_hub flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Receives/transmits over TCP to ESPHome devices' native API (port 6053)
  # and serves the Ingress web UI.
  network inet tcp,
  network inet6 tcp,

  # Data dir for the SQLite DB and the at-rest encryption key.
  /data/** rw,

  # Python runtime.
  /usr/bin/python3* ix,
  /usr/lib/python3.13/** r,
  /app/** r,

  # s6-overlay supervision.
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** rwix,
  /package/** ix,
  /command/** ix,
  /etc/s6-overlay/** rwix,

  deny /dev/tty[0-9]* rw,
}
